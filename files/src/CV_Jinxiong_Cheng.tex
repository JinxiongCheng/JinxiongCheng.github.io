\begin{points}
  \item Designed microfluidic platforms enabling high-throughput screening of
        \textbf{$\sim$3,000 compounds} at \textbf{single-cell} resolution.
  \item Quantified migration behavior across \textbf{$>$10,000 single cells} per experiment.
  \item Integrated \textbf{machine learning} with high-throughput screening to predict
        drug efficacy, enabling virtual screening of \textbf{$>$6,000 compounds}.
\end{points}

\rowbb{2. Suspension cell culture on digital microfluidics platform}{Toronto, Ontario, Canada}
\rowii{Visiting Research Student, Wheeler Lab, University of Toronto}{Sep 2022 - Dec 2022}

  \item Optimized droplet manipulation workflows achieving \textbf{$>$90\% cell viability}
        and high reproducibility across experiments.
\end{points}

\rowbb{3. dECM-based tumor-on-chip for drug screening}{Shenzhen, Guangdong, China}
\rowii{Undergraduate Researcher, Jiang Group, SUSTech}{Feb 2022 - Jun 2023}
\begin{points}
  \item Designed a high-throughput microfluidic platform integrating a
        \textbf{concentration gradient generator} and 3D tumor microenvironment.
  \item Generated stable linear drug gradients (\textbf{1--10 $\mu$L/min, 7 outlets})
        with maintained nanoparticle dispersion \textbf{over 7 days}.
  \item Established \textbf{vascularized co-culture systems}
        ($\sim$10\textsuperscript{6} cells/mL) for physiologically relevant drug screening.
\end{points}
\vspace{2.5pt}

\rowbb{4. Effects of nanoplastics on neuronal systems}{Shenzhen, Guangdong, China}
\rowii{Undergraduate Researcher, Jiang Group, SUSTech}{Sep 2021 - Jan 2022}
\begin{points}
  \item Designed a microfluidic \textbf{gradient chip} for controlled nanoparticle
        exposure and high-throughput screening.
  \item Quantified size- and ligand-dependent neuronal uptake of nanoplastics
        (80--200 nm).
  \item Demonstrated brain deposition and behavioral alterations \textbf{in vivo}
        following 7-day exposure.
\end{points}
\vspace{2.5pt}

\rowbb{5. Microfluidic system for multiplexed biomarker detection}{Shenzhen, Guangdong, China}
\rowii{Undergraduate Researcher, Jiang Group, SUSTech}{Mar 2021 - Apr 2022}
\begin{points}
  \item Characterized the intrinsic properties of PDMS films and evaluated their
        effects on the vitamin detection assays.
  \item Achieved high sensitivity (\textbf{LOD: 6.15 ng/mL}) and high SNR
        (\textbf{33.49 dB}).
  \item Enabled multiplexed detection with strong clinical correlation
        (R\textsuperscript{2} $>$ 0.9) at low cost ($\sim$\$0.04/test).
\end{points}
